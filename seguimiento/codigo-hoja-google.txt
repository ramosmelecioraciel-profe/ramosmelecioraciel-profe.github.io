const CLAVE_EN_CODIGO = '';       // opcional: si prefieres, escribe aquí tu clave en lugar de usar el menú

const HOJA_DATOS = '_datos';      // aquí se guarda todo (no la edites a mano)
const TROZO = 40000;              // caracteres por celda

function doPost(e) {
  let p = {};
  try { p = JSON.parse(e.postData.contents); } catch (x) {}
  return salida_(manejar_(p));
}
function doGet() {
  return salida_({ ok: true, mensaje: 'Servidor de seguimiento activo' });
}

function manejar_(p) {
  const clave = PropertiesService.getScriptProperties().getProperty('CLAVE') || CLAVE_EN_CODIGO;
  if (!clave) return { ok: false, error: 'sinclave' };
  if (String(p.clave || '') !== clave) return { ok: false, error: 'clave' };
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const cur = leer_();
    if (p.accion === 'cargar') return { ok: true, datos: cur.datos, rev: cur.rev };
    if (p.accion === 'guardar') {
      if (!p.datos || !Array.isArray(p.datos.grupos)) return { ok: false, error: 'datos' };
      if (Number(p.base) !== -1 && cur.rev && Number(p.base) !== cur.rev)
        return { ok: false, error: 'conflicto', datos: cur.datos, rev: cur.rev };
      const rev = cur.rev + 1;
      escribir_(p.datos, rev);
      return { ok: true, rev: rev };
    }
    return { ok: false, error: 'accion' };
  } finally {
    lock.releaseLock();
  }
}

function salida_(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

function hojaDatos_() {
  const ss = SpreadsheetApp.getActive();
  let h = ss.getSheetByName(HOJA_DATOS);
  if (!h) { h = ss.insertSheet(HOJA_DATOS); h.hideSheet(); }
  return h;
}
function leer_() {
  const h = hojaDatos_();
  const n = h.getLastRow();
  if (!n) return { datos: null, rev: 0 };
  const v = h.getRange(1, 1, n, 2).getValues();
  const rev = Number(v[0][1]) || 0;
  const txt = v.map(r => String(r[0]).replace(/^~/, '')).join('');
  if (!txt) return { datos: null, rev: rev };
  return { datos: JSON.parse(txt), rev: rev };
}
function escribir_(datos, rev) {
  const h = hojaDatos_();
  const txt = JSON.stringify(datos);
  const filas = [];
  for (let i = 0; i < txt.length; i += TROZO) filas.push(['~' + txt.slice(i, i + TROZO), '']);
  if (!filas.length) filas.push(['~', '']);
  filas[0][1] = rev;
  filas[0].push(new Date());
  for (let i = 1; i < filas.length; i++) filas[i].push('');
  h.clearContents();
  h.getRange(1, 1, filas.length, 3).setValues(filas);
}

function configurarClave() {
  const ui = SpreadsheetApp.getUi();
  const r = ui.prompt('Configurar mi clave', 'Escribe una clave (mínimo 6 caracteres). La usarás en la app junto con la dirección /exec.\nNo la compartas con tus alumnos.', ui.ButtonSet.OK_CANCEL);
  if (r.getSelectedButton() !== ui.Button.OK) return;
  const c = r.getResponseText().trim();
  if (c.length < 6) { ui.alert('La clave debe tener al menos 6 caracteres.'); return; }
  PropertiesService.getScriptProperties().setProperty('CLAVE', c);
  configurar();
  ui.alert('Listo. Ahora en Extensiones → Apps Script: Implementar → Nueva implementación → Aplicación web (Quién tiene acceso: Cualquier usuario) y copia la dirección /exec.');
}
function configurar() {
  ScriptApp.getProjectTriggers().filter(t => t.getHandlerFunction() === 'actualizarHojas')
    .forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('actualizarHojas').timeBased().everyHours(1).create();
  hojaDatos_();
  try { actualizarHojas(); } catch (e) { console.error(e); }
}
function onOpen() {
  SpreadsheetApp.getUi().createMenu('Seguimiento')
    .addItem('Configurar mi clave', 'configurarClave')
    .addItem('Actualizar resúmenes ahora', 'actualizarHojas').addToUi();
}

function actualizarHojas() {
  const d = leer_().datos;
  if (!d || !Array.isArray(d.grupos)) return;
  const ss = SpreadsheetApp.getActive();
  d.grupos.forEach(g => {
    try { resumenGrupo_(ss, g); } catch (e) { console.error('Resumen de ' + g.nombre + ': ' + e); }
  });
}
const MAT_ = { PM1: 'Pensamiento Matemático I', HC: 'Hoja de Cálculo Aplicada' };
function resumenGrupo_(ss, g) {
  g.alumnos = g.alumnos || []; g.acts = g.acts || []; g.asis = g.asis || {}; g.extra = g.extra || {};
  const nP = g.parciales || 3;
  const nombre = ('Resumen ' + (MAT_[g.materia] || g.materia) + ' ' + g.nombre).replace(/[\[\]\*\?\/\\:]/g, '-').slice(0, 90);
  const h = ss.getSheetByName(nombre) || ss.insertSheet(nombre);
  h.clear();
  const enc = ['Alumno', '% Asistencia', 'Faltas', 'Retardos', 'Justificadas'];
  for (let p = 1; p <= nP; p++) enc.push('Parcial ' + p);
  enc.push('Promedio', 'Alertas');
  const filas = [enc];
  g.alumnos.filter(a => !a.baja).sort((a, b) => a.nombre.localeCompare(b.nombre, 'es')).forEach(a => {
    const s = asis_(g, a.id);
    const f = [a.nombre, s.pct == null ? '' : Math.round(s.pct), s.F, s.R, s.J];
    const fins = [];
    for (let p = 1; p <= nP; p++) {
      const c = calif_(g, a.id, p);
      f.push(c.fin == null ? '' : (c.hayExam ? c.fin : Math.round(c.total * 10) / 10 + ' pts'));
      if (c.fin != null && c.hayExam) fins.push(c.fin);
    }
    f.push(fins.length ? Math.round(fins.reduce((x, y) => x + y, 0) / fins.length * 10) / 10 : '');
    f.push(alertas_(g, a.id).join(' · '));
    filas.push(f);
  });
  h.getRange(1, 1, filas.length, enc.length).setValues(filas);
  h.getRange(1, 1, 1, enc.length).setFontWeight('bold').setBackground('#1d3557').setFontColor('#ffffff');
  h.setFrozenRows(1);
  h.autoResizeColumns(1, enc.length);
}
/* --- mismos cálculos que la app --- */
const num_ = v => typeof v === 'number' && isFinite(v);
function parcialDe_(g, f) {
  const P = g.periodos || {}, n = g.parciales || 3; let p = 1;
  for (let q = 1; q <= n; q++) { const r = P[q] || {}; if (r.ini && r.fin && f >= r.ini && f <= r.fin) return q; }
  for (let q = 1; q <= n; q++) { const r = P[q] || {}; if (r.ini && f >= r.ini) p = q; }
  if (!g.periodos && g.inicios) for (let q = 2; q <= n; q++) if (g.inicios[q] && f >= g.inicios[q]) p = q;
  return p;
}
function asis_(g, aid, p) {
  let A = 0, F = 0, R = 0, J = 0;
  Object.keys(g.asis).forEach(f => {
    if (p && parcialDe_(g, f) !== p) return;
    const c = (g.asis[f].m || {})[aid];
    if (c === 'A') A++; else if (c === 'F') F++; else if (c === 'R') R++; else if (c === 'J') J++;
  });
  const tot = A + F + R + J;
  return { A, F, R, J, tot, pct: tot ? A / tot * 100 : null };
}
const pondA_ = (g, p) => num_((g.pond || {})[p]) ? g.pond[p] : 70;
const actMax_ = a => num_(a.max) && a.max > 0 ? a.max : 10;
const exMax_ = e => num_(e.max) ? e.max : 30;
function notaEf_(g, a, aid) {
  const v = (a.notas || {})[aid]; if (!num_(v)) return null;
  if (a.tipo === 'exam' || !a.tarde || !a.tarde[aid]) return v;
  const r = a.pen || (g.pen || {})[a.parcial] || { tipo: 'pct', v: 80 }, m = actMax_(a);
  if (r.tipo === 'pct') return v * r.v / 100;
  if (r.tipo === 'max') return Math.min(v, r.v / 10 * m);
  if (r.tipo === 'resta') return Math.max(0, v - r.v / 10 * m);
  return v;
}
function rubros_(g, p) {
  const R = (g.rub || {})[p];
  if (R && R.length) return R;
  const a0 = num_((g.pond || {})[p]) ? g.pond[p] : 70, p0 = num_((g.pondP || {})[p]) ? g.pondP[p] : 0;
  return [{ id: 'act', tipo: 'lista', peso: a0 }, { id: 'paec', tipo: 'unico', peso: p0 }, { id: 'exam', tipo: 'examen', peso: Math.max(0, 100 - a0 - p0) }];
}
const rubDe_ = a => a.tipo === 'exam' ? 'exam' : (a.rub || (a.tipo === 'paec' ? 'paec' : 'act'));
function calif_(g, aid, p) {
  const R = rubros_(g, p), ex = g.acts.find(a => a.parcial === p && a.tipo === 'exam');
  let total = 0, hay = false, hayEx = false, pend = 0;
  R.forEach(r => {
    const it = g.acts.filter(a => a.parcial === p && a.tipo !== 'exam' && rubDe_(a) === r.id);
    if (r.tipo === 'lista') {
      if (!it.length) return; hay = true;
      const fij = it.filter(a => num_(a.valor)), sumF = fij.reduce((s, a) => s + a.valor, 0), nAut = it.length - fij.length;
      const cada = nAut ? Math.max(0, r.peso - sumF) / nAut : 0;
      it.forEach(a => { const v = notaEf_(g, a, aid); if (num_(v)) total += v / actMax_(a) * (num_(a.valor) ? a.valor : cada); else pend++; });
    } else if (r.tipo === 'unico') {
      const a = it[0]; if (!a || !(r.peso > 0)) return; hay = true;
      const v = notaEf_(g, a, aid); if (num_(v)) total += v / actMax_(a) * r.peso; if (!num_((a.notas || {})[aid])) pend++;
    } else {
      if (!ex || !(r.peso > 0)) return; hay = true; hayEx = true;
      if (num_((ex.notas || {})[aid])) total += exMax_(ex) > 0 ? ex.notas[aid] / exMax_(ex) * r.peso : 0;
    }
  });
  const pe = (R.find(r => r.tipo === 'examen') || { peso: 0 }).peso;
  const extra = +(((g.extra || {})[p] || {})[aid]) || 0;
  let fin = null;
  if (hay) { const r = Math.round(total + extra * 10); fin = r < 60 ? 5 : Math.min(10, Math.round(r / 10)); }
  return { total: hay ? total : null, fin, hayExam: hayEx || (pe === 0 && hay), pend };
}
function alertas_(g, aid) {
  const o = [], a = asis_(g, aid);
  if (a.tot >= 3 && a.pct < 80) o.push('Asistencia ' + Math.round(a.pct) + '%');
  let n = 0;
  for (const f of Object.keys(g.asis).sort().reverse()) {
    const c = (g.asis[f].m || {})[aid]; if (!c || c === 'J') continue;
    if (c === 'F') n++; else break;
  }
  if (n >= 3) o.push(n + ' faltas seguidas');
  const hoy = Utilities.formatDate(new Date(), 'America/Mexico_City', 'yyyy-MM-dd');
  const p = parcialDe_(g, hoy), c = calif_(g, aid, p);
  if (c.pend >= 2) o.push(c.pend + ' actividades pendientes');
  if (c.hayExam && c.fin != null && c.fin < 6) o.push('Parcial ' + p + ': ' + c.fin);
  return o;
}
